// Read pH sensor and transmit data
const int PH_PIN = A0;

float calibrationSlope = -5.70;
float calibrationIntercept = 21.34;

void setup() {
  Serial.begin(115200);
}

void loop() {
  int raw = analogRead(PH_PIN);

  float voltage =
    raw * 5.0 / 1023.0;

  float pH =
    calibrationSlope * voltage
    + calibrationIntercept;

  Serial.print("PH=");
  Serial.println(pH, 2);

  delay(1000);
}
